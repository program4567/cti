3. NETWORK THREAT DETECTION USING PYSHARK
-----------------------------------------

import pyshark

capture = pyshark.FileCapture("capture.pcapng")

syn = 0
icmp = 0
ports = set()

for p in capture:
    try:
        if 'TCP' in p:
            if p.tcp.flags_syn == '1' and p.tcp.flags_ack == '0':
                syn += 1
                ports.add(p.tcp.dstport)

        if 'ICMP' in p:
            icmp += 1
    except:
        pass

capture.close()

print("SYN Packets:", syn)
print("ICMP Packets:", icmp)
print("Ports:", len(ports))

if syn > 20:
    print("Possible SYN Flood")

if len(ports) > 15:
    print("Possible Port Scan")

if icmp > 30:
    print("Possible ICMP Flood")

if syn <= 20 and len(ports) <= 15 and icmp <= 30:
    print("No major threat")

import hashlib
import re

file = "suspicious_sample.bin"

with open(file, "rb") as f:
    data = f.read()

md5 = hashlib.md5(data).hexdigest()
sha256 = hashlib.sha256(data).hexdigest()
strings = re.findall(b"[ -~]{4,}", data)

print("File Size:", len(data))
print("MD5:", md5)
print("SHA256:", sha256)

print("\nStrings Found:")
for s in strings:
    print(s.decode())

9. NETWORK BEHAVIOR ANALYSIS
----------------------------

import csv
from collections import Counter

ips = Counter()
ports = Counter()
protocols = Counter()

with open("malware_traffic.csv") as f:
    for row in csv.DictReader(f):
        ips[row["Destination"]] += 1
        ports[row["Destination Port"]] += 1
        protocols[row["Protocol"]] += 1

print("IPs:", ips.most_common(5))
print("Ports:", ports.most_common(5))
print("Protocols:", protocols)
